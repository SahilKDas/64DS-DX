cmake_minimum_required(VERSION 3.15)
project(client_pack_probe C CXX)
set(CMAKE_CXX_STANDARD 17)
set(PACK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
file(GLOB LUA_SOURCES "${PACK_ROOT}/vendor/lua/src/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
add_library(pack_lua STATIC ${LUA_SOURCES})
target_include_directories(pack_lua PUBLIC "${PACK_ROOT}/vendor/lua/src")
add_executable(resource_pack_probe ../resource_pack_probe.cpp "${PACK_ROOT}/hal/resource_pack.cpp")
target_include_directories(resource_pack_probe PRIVATE "${PACK_ROOT}/hal")
target_link_libraries(resource_pack_probe PRIVATE pack_lua)
